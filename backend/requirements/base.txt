# =============================================================================

# Django core
# 5.2 is the current LTS line (security support to April 2028).
# 5.1.x is end-of-life and carries ~30 known CVEs - do NOT downgrade.

# DRF 3.18.x is the first line that requires Django>=5.2
djangorestframework==3.18.3
django-cors-headers==4.9.0

# Database

# Redis (cache + event bus)

redis==8.1.0

# Authentication
djangorestframework-simplejwt==5.5.1
PyJWT==2.15.1

# Password hashing (Argon2id - memory-hard, GPU-attack resistant)
argon2-cffi==25.1.0

# Celery (task queue: EOD reports, data checks)
celery==5.6.3
django-celery-beat==2.9.0

# Configuration from environment variables
python-decouple==3.8

# Input sanitization (output encoding / HTML stripping)

# Security headers - Content-Security-Policy middleware

# HTTP client (for broker APIs in Phase 1)

# Async server + market data websockets (Phase 1)

# Monitoring / error tracking
